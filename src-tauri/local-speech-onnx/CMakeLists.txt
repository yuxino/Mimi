cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(MimiLocalOnnx LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mimi-local-onnx main.cpp)
target_include_directories(mimi-local-onnx PRIVATE "${SHERPA_ROOT}/include")
find_library(SHERPA_C_API NAMES sherpa-onnx-c-api PATHS "${SHERPA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(mimi-local-onnx PRIVATE "${SHERPA_C_API}")
if(APPLE)
  set_target_properties(mimi-local-onnx PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(mimi-local-onnx PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
if(BUILD_TESTING)
  add_executable(mimi-onnx-worker-test worker-test.cpp)
  target_include_directories(mimi-onnx-worker-test PRIVATE "${SHERPA_ROOT}/include")
  target_link_libraries(mimi-onnx-worker-test PRIVATE "${SHERPA_C_API}")
  add_test(NAME bounded-worker-protocol COMMAND mimi-onnx-worker-test)
endif()
if(MSVC)
  target_compile_options(mimi-local-onnx PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(mimi-local-onnx PRIVATE -Wall -Wextra -Werror)
endif()
